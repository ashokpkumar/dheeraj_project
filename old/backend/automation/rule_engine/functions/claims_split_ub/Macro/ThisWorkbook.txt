Private Sub Workbook_Open()
    MACRO_LIST_AND_LOG_SETUP
    Force_Excel_ToNew_Instance True
    With ThisWorkbook
        Select Case .Worksheets("Main").OLEObjects("chkDetails").Object.Value
            Case True
                .Worksheets("ClaimInfo").Visible = -1
                .Worksheets("ClaimServiceLines").Visible = -1
            Case False
                .Worksheets("ClaimInfo").Visible = 2
                .Worksheets("ClaimServiceLines").Visible = 2
        End Select
    End With
End Sub
