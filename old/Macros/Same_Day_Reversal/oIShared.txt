Public objSession As Object

Public Function Current_Session(ByVal SessionPath As String) As Boolean
    Dim objSystem As Object, objSessions As Object, msg As String
    On Error GoTo ErrHandler
    Current_Session = False
    Set objSystem = CreateObject("EXTRA.System")
    If (objSystem Is Nothing) Then
        msg = "Could not create the EXTRA System object.  Stopping macro playback."
    End If
    Set objSessions = objSystem.Sessions
    If (objSessions Is Nothing) Then
        msg = "Could not create the Sessions collection object.  Stopping macro playback."
    End If
    OldSystemTimeout& = objSystem.TimeoutValue: g_HostSettleTime = 500
    OldSystemTimeout& = objSystem.TimeoutValue
    If (g_HostSettleTime > OldSystemTimeout) Then objSystem.TimeoutValue = g_HostSettleTime
    If Len(Trim(SessionPath)) > 0 Then
        Set objSession = objSessions.Item(Trim(SessionPath))
    Else
        Set objSession = objSystem.ActiveSession
    End If
    If (objSession Is Nothing) Then
        msg = "Could not create the Session object.  Stopping macro playback."
    End If
    Current_Session = True
    Exit Function
ErrHandler:
    MsgBox "Unable to communicate with session." & _
    vbNewLine & vbNewLine & msg & vbNewLine & vbNewLine & _
    Err.Description, vbCritical, ThisWorkbook.Name
End Function

Public Function PlaceValue(ByVal Val As String, ByVal r As Integer, c As Integer)
    If Len(Trim(Val)) < 0 Then Exit Function
    With objSession.Screen
        .moveTo r, c
         Do While .OIA.Xstatus <> 0: DoEvents: Loop
        .SendKeys ("<EraseEof>")
         Do While .OIA.Xstatus <> 0: DoEvents: Loop
        .PutString Trim(Val), r, c
         Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Function

Public Function InCorrectScreen(ByVal ScrnVal As String, ByVal r As Integer, ByVal c As Integer, ByVal L As Integer) As Boolean
    Dim Scrn As String: InCorrectScreen = False
    With objSession.Screen
        Do While .OIA.Xstatus <> 0: DoEvents: Loop
        Scrn = Trim(.GetString(r, c, L))
        Select Case Scrn
            Case ScrnVal
                InCorrectScreen = True
        End Select
    End With
End Function
