Public Function UB_TOS_Entry(ByVal newTos As String, ByVal oldTos As String, ByVal Edit As String, ByVal rw As Long, ByRef mw As Worksheet)
    With objSess.Screen
        Select Case Mid(Edit, 6, 1)
            Case "1"
                PlaceValue CStr(newTos), 6, 14
                If (Trim(.GetString(7, 40, 11)) = ".00" Or Trim(.GetString(7, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 7, 14
                If (Trim(.GetString(8, 40, 11)) = ".00" Or Trim(.GetString(8, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 8, 14
                If (Trim(.GetString(9, 40, 11)) = ".00" Or Trim(.GetString(9, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 9, 14
            Case "2"
                PlaceValue CStr(newTos), 7, 14
                If (Trim(.GetString(6, 40, 11)) = ".00" Or Trim(.GetString(6, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 6, 14
                If (Trim(.GetString(8, 40, 11)) = ".00" Or Trim(.GetString(8, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 8, 14
                If (Trim(.GetString(9, 40, 11)) = ".00" Or Trim(.GetString(9, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 9, 14
            Case "3"
                PlaceValue CStr(newTos), 8, 14
                If (Trim(.GetString(6, 40, 11)) = ".00" Or Trim(.GetString(6, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 6, 14
                If (Trim(.GetString(7, 22, 10)) = ".00" Or Trim(.GetString(7, 22, 10)) = ".01") Then PlaceValue CStr(newTos), 7, 14
                If (Trim(.GetString(9, 40, 11)) = ".00" Or Trim(.GetString(9, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 9, 14
            Case "4"
                PlaceValue CStr(newTos), 9, 14
                If (Trim(.GetString(6, 40, 11)) = ".00" Or Trim(.GetString(6, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 6, 14
                If (Trim(.GetString(7, 22, 10)) = ".00" Or Trim(.GetString(7, 22, 10)) = ".01") Then PlaceValue CStr(newTos), 7, 14
                If (Trim(.GetString(8, 40, 11)) = ".00" Or Trim(.GetString(8, 40, 11)) = ".01") Then PlaceValue CStr(newTos), 8, 14
        End Select
       'Update 02/08 - Replace if there is 947 with 908
        If Len(Trim(.GetString(12, 14, 3))) > 0 And Trim(.GetString(12, 14, 3)) = Trim(oldTos) Then PlaceValue Trim(Tos), 12, 14
        If Len(Trim(.GetString(13, 14, 3))) > 0 And Trim(.GetString(13, 14, 3)) = Trim(oldTos) Then PlaceValue Trim(Tos), 13, 14
        If Len(Trim(.GetString(14, 14, 3))) > 0 And Trim(.GetString(14, 14, 3)) = Trim(oldTos) Then PlaceValue Trim(Tos), 14, 14
        If Len(Trim(.GetString(15, 14, 3))) > 0 And Trim(.GetString(15, 14, 3)) = Trim(oldTos) Then PlaceValue Trim(Tos), 15, 14
        PlaceValue "x", 29, 71
       .SendKeys "<ENTER>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Function
