Public Function CURRENT_SESSION() As Boolean
    Dim objSystem As Object, objSessions As Object, msg As String
    On Error GoTo ErrHandler
    CURRENT_SESSION = False
    Set objSystem = CreateObject("EXTRA.System")
    If (objSystem Is Nothing) Then
        msg = "Could not create the EXTRA System object.  Stopping macro playback."
    End If
    Set objSessions = objSystem.Sessions
    If (objSessions Is Nothing) Then
        msg = "Could not create the Sessions collection object.  Stopping macro playback."
    End If
    OldSystemTimeout& = objSystem.TimeoutValue: g_HostSettleTime = 500
    OldSystemTimeout& = objSystem.TimeoutValue
    If (g_HostSettleTime > OldSystemTimeout) Then objSystem.TimeoutValue = g_HostSettleTime
    If Len(Trim(MN.Range("B3"))) > 0 Then
        Set objSession = objSessions.Item(Trim(MN.Range("B3")))
    Else
        Set objSession = objSystem.ActiveSession
    End If
    If (objSession Is Nothing) Then
        msg = "Could not create the Session object.  Stopping macro playback."
    End If
    CURRENT_SESSION = True
    Exit Function
ErrHandler:
    MsgBox "Unable to communicate with session." & _
    vbNewLine & vbNewLine & msg & vbNewLine & vbNewLine & _
    Err.Description, vbCritical, ThisWorkbook.Name
End Function

Public Function PLACEVALUE(ByVal val As String, ByVal R As Integer, c As Integer)
    If Len(Trim(val)) < 1 Then Exit Function
    With objSession.screen
        .moveTo R, c
        .SendKeys ("<EraseEof>")
        .PutString Trim(val), R, c
         Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Function

Public Function REMOVEVALUE(ByVal R As Integer, c As Integer) 'Added 2022.07.20
    With objSession.screen
        .moveTo R, c
        .SendKeys ("<EraseEof>")
         Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Function

Public Function INCORRECTSCREEN(ByVal ScrnVal As String, ByVal R As Integer, ByVal c As Integer, ByVal L As Integer) As Boolean
    Dim Scrn As String: INCORRECTSCREEN = False
    With objSession.screen
        Do While .OIA.Xstatus <> 0: DoEvents: Loop
        Scrn = Trim(.GetString(R, c, L))
        Select Case Scrn
            Case ScrnVal
                INCORRECTSCREEN = True
        End Select
    End With
End Function

Public Function REFORMAT_CELL(ByRef wsRef As Worksheet, ByVal Target As String)
    With wsRef
        .Range(Target).NumberFormat = "@"
        .Range(Target) = "'" & Trim(.Range(Target).Value)
         With .Range(Target).Font
            .Name = "Calibri": .Size = 8: .Strikethrough = False
            .Superscript = False: .Subscript = False
            .OutlineFont = False: .Shadow = False
            .Underline = xlUnderlineStyleNone
         End With
    End With
End Function

Public Function NORMALIZE_EDIT_MSG(ByVal str As String) As String
    Dim i As Integer
    For i = 0 To UBound(Split(str, " "))
        If Len(Trim(Split(str, " ")(i))) > 0 Then NORMALIZE_EDIT_MSG = NORMALIZE_EDIT_MSG & Trim(Split(str, " ")(i)) & " "
    Next i
    NORMALIZE_EDIT_MSG = UCase(Trim(NORMALIZE_EDIT_MSG))
End Function

Public Function Force_Excel_ToNew_Instance(ByVal bln As Boolean)
    Dim ws As New WshShell, sKey As String
    sKey = "HKEY_CURRENT_USER\Software\Microsoft\Office\16.0\Excel\Options\DisableMergeInstance"
    If RegKeyExists(sKey, KeyVal) = True Then
        If KeyVal <> 1 Then
            ws.RegDelete sKey
            GoTo SaveKey
        End If
    Else
SaveKey:
        ws.RegWrite sKey, 1, "REG_DWORD"
    End If
    Set ws = Nothing
End Function

Private Function RegKeyExists(ByVal sKey As String, KeyVal) As Boolean
    Dim ws As WshShell
    On Error GoTo ErrHandler
    Set ws = New WshShell
    KeyVal = ws.RegRead(sKey)
    RegKeyExists = True
Rel_Obj:
    Set ws = Nothing
    Exit Function
ErrHandler:
    RegKeyExists = False
    GoTo Rel_Obj
End Function
