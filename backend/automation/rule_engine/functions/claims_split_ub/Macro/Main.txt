Private Sub chkDetails_Click()
    Select Case chkDetails.Value
        Case True
            CI.Visible = -1: CS.Visible = -1
        Case False
            CI.Visible = 2: CS.Visible = 2
    End Select
End Sub

Private Sub chkWbClaim_Click()
    If chkWbClaim.Value <> False Then
        If MsgBox("USE BELOW RESOURCE URL" & vbCrLf & vbCrLf & wURL_DOMAIN & vbCrLf & _
        vbCrLf & "Do you want to continue?", _
        vbQuestion + vbYesNo, ThisWorkbook.Name) <> vbYes Then _
        chkWbClaim.Value = False Else chkWbClaim.Value = True
    End If
End Sub

Private Sub cmdRUN_Click()
    Dim MacroAssetName As String, MacroStartTime As Date, MacroEndTime As Date, MacroFile As String, MacroTransaction As String, MacroAssetId As String, MacroComments As String
    Local_Folder = Environ("LocalAppData") & "\Macros\VBS"
    AAI_ID = AAI_ASSET_INFO(Local_Folder & "\", fASSETLIST)
    
   'AAI LOG ATTRIBUTES
    MacroAssetName = mASSETNAME 'OPTIONAL (ORIGINAL AAI REGISTRATION)
    MacroFile = "" 'OPTIONAL (ORIGINAL REGISTERED MACRO FILE)
    MacroComments = mCURRENTVR & "|" & ThisWorkbook.Name 'OPTIONAL (USER RENAMEED FILE)
    MacroStartTime = Now() 'REQUIRED
    MacroAssetId = AAI_ID 'REQUIRED
    tTrans = 0
   '**************************
   'START OF AUTOMATION
   '**************************
    Dim x As Long, y As Long, pFile As String, rW As Long, tClaim As String, CCN As String, aCTION As String
    Dim TotalCharges As Double, TotalRepriced, TotalDiscounts, Re_Priced_Ind, HIC_Number, Re_Priced_By, TimelyFiling, MethodInfo, ClaimNTE
    With MN
        aCTION = Trim(.Range("I1"))
        If MsgBox("ACTION TO PERFORM: " & vbCrLf & aCTION & vbCrLf & vbCrLf & _
        "Do you want to continue?", vbQuestion + vbYesNo, "HELPER") <> vbYes Then Exit Sub
        Select Case aCTION
            Case "01.CLEAN ALL SHEETS"
                CLEAN_ALL_WORKSHEETS
               .chkImg.Value = True
               .chkWbClaim.Value = False
            Case "02.GET EDI DETAILS"
                fnLOC = Environ("temp")
                For rW = .Range("F1") To .Range("F2")
                   .Range("G2") = rW: .Range("E3") = "MACRO IS CURRENTLY RUNNING…."
                    Set iREADER = CreateObject("PdfClaimImageDetails.ReadClaimDetails")
                    REFORMAT_CELL MN, .Range("C" & rW).Address
                    CCN = Trim(MN.Range("C" & rW))
                    If Len(CCN) <> 11 Then GoTo NXTCLAIM
                    pFile = fnLOC & "\" & CCN & ".pdf"
                    If .chkWbClaim.Value <> True Then tClaim = Get_PDF_Claim_New(CCN) Else tClaim = NEW_WEBLCLAIM(CCN) 'ADJUSTED 2025.11.04
                    TotalCharges = 0: TotalRepriced = "": TotalDiscounts = "": TimelyFiling = "": MethodInfo = "": ClaimNTE = ""
                    Select Case tClaim
                        Case "UB"
                           .Range("D" & rW) = tClaim
                            If File_Exists(pFile) <> True Then
                                .Range("A" & rW) = "FILE NOT EXISTS": GoTo NXTCLAIM
                            End If
                            x = CI.Cells(CI.Rows.Count, 1).End(xlUp).Row + 1
                            CLAIM_DEMOGRAPHICS_INFORMATION pFile, x, CCN
                            y = CS.Cells(CS.Rows.Count, 1).End(xlUp).Row + 1
                            CLAIM_SERVICELINES_INFORMATION pFile, y, CCN, TotalCharges
                            CI.Range("HB" & x) = "'" & TotalCharges
                            CLAIM_REPRICING_INFORMATION pFile, y, TotalRepriced, TotalDiscounts, Re_Priced_Ind, HIC_Number, Re_Priced_By, TimelyFiling, MethodInfo, ClaimNTE
                            If TotalRepriced = "" Then TotalRepriced = "-"
                            CI.Range("HC" & x) = "'" & Format(TotalRepriced, "0.00")
                            If TotalDiscounts = "" Then TotalDiscounts = "-"
                            CI.Range("HD" & x) = "'" & Format(TotalDiscounts, "0.00")
                            CI.Range("HE" & x) = "'" & Re_Priced_Ind
                            CI.Range("HO" & x) = "'" & HIC_Number
                            CI.Range("HQ" & x) = "'" & Re_Priced_By
                            CI.Range("HS" & x) = "'" & MethodInfo
                            CI.Range("HT" & x) = "'" & TimelyFiling
                            CI.Range("HU" & x) = "'" & ClaimNTE
                        Case "Not Found.", "CCN Missing.", ""
                           .Range("A" & rW) = "CANCELLED"
                           .Range("B" & rW) = "UNABLE TO VEIW CLAIM": GoTo NXTCLAIM
                        Case "HCFA"
                           .Range("A" & rW) = "CANCELLED"
                           .Range("B" & rW) = "CLAIM NOT SUPPORTED BY MACRO.": GoTo NXTCLAIM
                        Case Else
                           .Range("A" & rW) = "CANCELLED"
                           .Range("B" & rW) = tClaim
                    End Select: DoEvents
                    Set iREADER = Nothing
                    DEL_EXISTING_FILE fnLOC & "\" & CCN & ".pdf"
NXTCLAIM:
                    DoEvents
                Next rW
                Populate_Main_Sheet True
            Case "03.SPLIT CLAIM"
                If CURRENT_SESSION <> True Then Exit Sub
                For rW = .Range("F1") To .Range("F2")
                   .Range("E3") = "MACRO IS CURRENTLY RUNNING…."
                    CCN = Trim(MN.Range("C" & rW))
                    If Len(CCN) <> 11 Then GoTo NXTROW
                    If .Range("D" & rW) <> "UB" Then GoTo NXTROW
                    Select Case .Range("X2")
                        Case "SCRATCH"
                            If Len(Trim(.Range("P" & rW))) < 1 Then
                                .Range("A" & rW) = "CANCELLED.": .Range("B" & rW) = "NEW DOS IS REQUIRED.": GoTo NXTROW
                            End If
                            UB_Scratch_Split Trim(.Range("C" & rW)), Trim(.Range("O" & rW)), Trim(.Range("N" & rW)), Format(.Range("P" & rW), "MMDDYY"), rW
                        Case "SCRATCH NOT ONLINE"
                            If Len(Trim(.Range("P" & rW))) < 1 Then
                                .Range("A" & rW) = "CANCELLED.": .Range("B" & rW) = "NEW DOS IS REQUIRED.": GoTo NXTROW
                            End If
                            UB_Scratch_NotOnline (.Range("C" & rW)), Trim(.Range("O" & rW)), Trim(.Range("N" & rW)), Format(.Range("P" & rW), "MMDDYY"), rW
                        Case "NON SCRATCH"
                            UB_NonScratch_Split rW
                    End Select
NXTROW:
                    DoEvents
                Next rW
        End Select
       .Range("E3").ClearContents
    End With
   '**************************
    MacroEndTime = Now() 'REQUIRED
    MacroTransaction = IIf(tTrans > 0, CStr(tTrans), "") 'OPTIONAL (TOTAL TRANSACTION)
    If Len(Trim(MacroAssetId)) > 0 Then
        Call Usage_Log_AAI(MacroAssetName, MacroStartTime, MacroEndTime, MacroFile, MacroTransaction, MacroAssetId, MacroComments, AutomationTeam, AAIEnvironment)
    End If
    MsgBox "MACRO IS FINISHED", vbInformation, ThisWorkbook.Name
End Sub

Private Sub Worksheet_Change(ByVal Target As Range)
    With MN
        Select Case Target.Address
            Case "$I$1"
               .chkWbClaim.Visible = False
               .chkWbClaim.Value = False
               .chkImg.Visible = False
               .chkImg.Value = True
                Select Case UCase(Target.Value)
                    Case "01.CLEAN ALL SHEETS", "03.SPLIT CLAIM"
                    Case "02.GET EDI DETAILS"
                        .chkImg.Visible = True: .chkWbClaim.Visible = True
                End Select
            Case "$X$2"
                Select Case UCase(Target.Value)
                    Case "SCRATCH"
                        Init_Scratch_Controls True
                       .chkCPSDisc.Visible = False 'MN.OLEObjects("chkCPSDisc").Visible = False
                    Case "NON SCRATCH"
                        Init_Scratch_Controls False
                       .chkCPSDisc.Visible = True 'MN.OLEObjects("chkCPSDisc").Visible = True
                End Select
        End Select
    End With
End Sub

Private Sub Init_Scratch_Controls(ByVal bln As Boolean)
    With MN
        Select Case bln
            Case True
                .Columns("N:R").Hidden = False
            Case False
                .Columns("N:R").Hidden = True
        End Select
    End With
End Sub
