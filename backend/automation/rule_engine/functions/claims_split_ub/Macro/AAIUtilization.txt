Public Function AAI_ASSET_INFO(ByVal pDir As String, ByVal fName As String) As String
    Dim objCon As ADODB.Connection, oConStr As String, objRs As ADODB.Recordset, QryStr As String
    oConStr = "Provider=Microsoft.ACE.OLEDB.12.0;Data Source=" & pDir & ";" & _
    "Extended Properties=" & """" & "Text;HDR=Yes;FMT=Delimited;ReadOnly=True" & """" & ";"
    AAI_ASSET_INFO = ""
    AAI_ASSET_INI pDir, fName
    Set objCon = New ADODB.Connection
    If objCon.State = adStateClosed Then objCon.Open oConStr
    QryStr = "SELECT TOP 1 [ASSETID] FROM [" & fName & "] WHERE [ASSET NAME] = '" & mASSETNAME & _
    "' AND UCASE([ASSET STATUS]) = 'ACTIVE' ORDER BY CDATE([CREATIONDATE]) DESC;"
    Set objRs = New ADODB.Recordset
    objRs.Open QryStr, objCon, adOpenKeyset, adLockReadOnly, adCmdText
    If Not objRs.EOF Then AAI_ASSET_INFO = objRs(0)
    If objRs.State Then objRs.Close
    If objCon.State = adStateOpen Then objCon.Close
    DEL_EXISTING_FILE Local_Folder & "\" & "Schema.ini"
End Function

Public Function DEL_EXISTING_FILE(ByVal Killfile As String) As String
     DEL_EXISTING_FILE = "NOT DELETED"
     If Len(Dir$(Killfile)) > 0 Then
         SetAttr Killfile, vbNormal: Kill Killfile
         DEL_EXISTING_FILE = "DELETED"
     End If
End Function

Public Sub Create_Folders(ByVal folderPath As String)  'AAI SCRIPT
    Dim FileSys As FileSystemObject
    Set FileSys = New FileSystemObject
    For Each Folder In Split(folderPath, "\")
        Create_Folder = Create_Folder & Folder & "\"
        If Not FileSys.FolderExists(Create_Folder) Then
            On Error Resume Next
            FileSys.CreateFolder (Create_Folder)
        End If
    Next
    Set FileSys = Nothing
End Sub

Public Function File_ModDate_Compare(ByVal origFile As String, ByVal destFile As String) As Boolean   'AAI SCRIPT
    Dim FileSys As FileSystemObject
    File_ModDate_Compare = False
    Set FileSys = New FileSystemObject
    If (Modified_File_Check(origFile) <> Modified_File_Check(destFile)) Or _
    Not FileSys.FileExists(destFile) Then
        File_ModDate_Compare = True
    End If
    Set FileSys = Nothing
End Function

Public Function Modified_File_Check(ByVal filePath As String) As Date   'AAI SCRIPT
    Dim FileSys As FileSystemObject
    Set FileSys = New FileSystemObject
    If FileSys.FileExists(filePath) Then
        Set modFile = FileSys.GetFile(filePath)
        Modified_File_Check = Trim(modFile.DateLastModified)
    Else
        Modified_File_Check = "1/1/1899"
    End If
    Set FileSys = Nothing: Set modFile = Nothing
End Function

Public Sub Copy_File(ByVal origFile As String, ByVal destFile As String) 'AAI SCRIPT
    Dim FileSys As FileSystemObject
    Set FileSys = New FileSystemObject
    If FileSys.FileExists(origFile) Then
        On Error Resume Next
        FileSys.CopyFile origFile, destFile, True
    End If
    Set FileSys = Nothing
End Sub

Public Function File_Exists(ByVal filePath As String) As Boolean  'ADDED 2025.05.02
    Dim FileSys As FileSystemObject
    Set FileSys = New FileSystemObject
    File_Exists = False
    If FileSys.FileExists(filePath) Then File_Exists = True
    Set FileSys = Nothing
End Function

Public Sub RegisterDLL(DLLPath As String)
   'Path to RegAsm, adjust to the version of the .Net framework you're targeting
    Dim RegasmPath As String, LibraryPath As String, strNewRegPath As String, ws As WshShell, strRegFileContent As String, fileNo As Integer
    RegasmPath = "C:\Windows\Microsoft.NET\Framework\" & Dir("C:\Windows\Microsoft.NET\Framework\v4*", vbDirectory) & "\RegAsm.exe"
    Set ws = New WshShell
    #If Win64 Then
        Const Win64 = True
    #Else
        Const Win64 = False
    #End If
    LibraryPath = Left(DLLPath, Len(DLLPath) - 4)
    If Dir(DLLPath) = "" Then 'Check if file exists
        Err.Raise 1, Description:="Missing DLL!"
        Exit Sub
    End If
    ws.Run """" & RegasmPath & """ """ & DLLPath & """ /codebase /regfile", 0, True 'Create .reg file using RegAsm
    If Not Win64 And Environ$("ProgramW6432") <> vbNullString Then
        strNewRegPath = "HKEY_CURRENT_USER\SOFTWARE\Classes\Wow6432Node" '32 bits Office on Win 64
    Else
        strNewRegPath = "HKEY_CURRENT_USER\SOFTWARE\Classes" 'Default registry path
    End If
    fileNo = FreeFile
    Open LibraryPath & ".reg" For Binary Access Read Write As #fileNo
        strRegFileContent = String(LOF(fileNo), vbNullChar)
        Get #fileNo, , strRegFileContent 'Read reg contents
        strRegFileContent = Replace(strRegFileContent, "HKEY_CLASSES_ROOT", strNewRegPath) 'Change reg path
        Put #fileNo, 1, strRegFileContent 'Overwrite, always extends so no need to truncate
    Close #fileNo
    ws.Run "regedit.exe /s """ & LibraryPath & ".reg""", 0, True  'Merge silently into registry
    Kill LibraryPath & ".reg" 'Clean up registry
    Set ws = Nothing
End Sub

Public Sub UnregisterDLL(DLLPath As String)
    Dim RegasmPath As String, LibraryPath As String, strNewRegPath As String, ws As WshShell, strRegFileContent As String, fileNo As Integer, fileOutput As Integer
    RegasmPath = "C:\Windows\Microsoft.NET\Framework\" & Dir("C:\Windows\Microsoft.NET\Framework\v4*", vbDirectory) & "\RegAsm.exe"
    Set ws = New WshShell
    #If Win64 Then
        Const Win64 = True
    #Else
        Const Win64 = False
    #End If
    LibraryPath = Left(DLLPath, Len(DLLPath) - 4)
    If Dir(DLLPath) = "" Then 'Check if file exists
        Err.Raise 1, Description:="Missing DLL!"
        Exit Sub
    End If
    ws.Run """" & RegasmPath & """ """ & DLLPath & """ /codebase /regfile", 0, True 'Create .reg file using RegAsm
    If Not Win64 And Environ$("ProgramW6432") <> vbNullString Then
        strNewRegPath = "HKEY_CURRENT_USER\SOFTWARE\Classes\Wow6432Node" '32 bits Office on Win 64
    Else
        strNewRegPath = "HKEY_CURRENT_USER\SOFTWARE\Classes" 'Default registry path
    End If
    fileNo = FreeFile
    fileOutput = FreeFile + 1
    Open LibraryPath & ".reg" For Input As #fileNo
    Open LibraryPath & "remove.reg" For Output As #fileOutput
        Line Input #fileNo, strRegFileContent 'Read reg contents
        Print #fileOutput, strRegFileContent 'Copy first line blindly
        Do While Not EOF(fileNo)
            Line Input #fileNo, strRegFileContent 'Read reg contents
            strRegFileContent = Replace(strRegFileContent, "HKEY_CLASSES_ROOT", strNewRegPath) 'Change reg path
            If Left(strRegFileContent, 1) = "[" Then 'If new key
                Print #fileOutput, "[-" & Mid(strRegFileContent, 2) 'Change to remove key
            End If
        Loop
    Close #fileOutput
    Close #fileNo
    Kill LibraryPath & ".reg" 'Remove create file
    ws.Run "regedit.exe /s """ & LibraryPath & "remove.reg"""  'Merge silently into registry
    Kill LibraryPath & "remove.reg" 'Remove delete file
    Set ws = Nothing
End Sub
