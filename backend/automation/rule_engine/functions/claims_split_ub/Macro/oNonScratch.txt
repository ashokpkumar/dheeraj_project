Public Function ClaimPgCount(ByVal CCN As String) As Integer ' ADDED 2023.08.03
    Dim PgCnt As Integer
    ClaimPgCount = 0
    With objSession.screen
ReCountPg:
        ClaimPgCount = ClaimPgCount + 1
        For i = 6 To 18 Step 2
            If Len(Trim(.Getstring(i, 2, 2))) < 1 Then Exit For
        Next i
        If InStr(1, Trim(.Getstring(20, 2, 60)), "MORE DATA", vbTextCompare) > 0 Then
            .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
            GoTo ReCountPg
        End If
        .SendKeys ("<Pf9>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
         PLACEVALUE CCN, 8, 15
        .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Function

Public Function GetCPS_Discounts(ByVal CCN As String) As Integer  'Added 2023.08.01
    Dim r As Integer, i As Integer, ln As String, wS As Worksheet, rng As Range, lstRw As Long, l As Range, PgCnt As Integer, Pg As Integer, p As Integer
    Set wS = ThisWorkbook.Worksheets("Main"):
    If wS.AutoFilterMode Then wS.AutoFilterMode = False
    wS.Application.ScreenUpdating = False
    lstRw = wS.Cells(wS.Rows.Count, "S").End(xlUp).Row
    With objSession.screen
        PgCnt = ClaimPgCount(CCN): p = 1
ReCountDraft:
        For Pg = 1 To PgCnt ' ADDED 2023.08.03
            Do While p < Pg ' ADDED 2023.08.03
               .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop: p = p + 1 ' ADDED 2023.08.03
                DoEvents ' ADDED 2023.08.03
            Loop
            For r = 6 To 18 Step 2
                ln = Trim(.Getstring(r, 2, 2))
                If Len(ln) > 0 Then
                    PLACEVALUE ln, 3, 26
                   .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
                   .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
                    For i = 6 To 9
                        If Len(Trim(.Getstring(i, 3, 4))) > 0 Then
                           'If Not colTOS.Exists(Trim(.getstring(i, 3, 4))) Then colTOS.Add Trim(.getstring(i, 3, 4)), Trim(.getstring(i, 14, 2))
                            wS.Range("$S$5:$AE" & lstRw).AutoFilter Field:=1, Criteria1:=CCN
                            wS.Range("$S$5:$AE" & lstRw).AutoFilter Field:=3, Criteria1:=Trim(.Getstring(i, 3, 4)) 'REVCODE
                            wS.Range("$S$5:$AE" & lstRw).AutoFilter Field:=4, Criteria1:=Trim(.Getstring(i, 8, 5)) 'HCPCS/CPT
                            wS.Range("$S$5:$AE" & lstRw).AutoFilter Field:=2, Criteria1:=Left(Trim(.Getstring(i, 17, 6)), 2) & "/" & Mid(Trim(.Getstring(i, 17, 6)), 3, 2) & "/" & Right(Trim(.Getstring(i, 17, 6)), 2) 'SERDT
                            If wS.AutoFilter.Range.Columns("S").SpecialCells(xlCellTypeVisible).Cells.Count - 1 > 0 Then
                                Set rng = wS.AutoFilter.Range.Columns(5).Offset(1, 0).Resize(lstRw - 5, 1).SpecialCells(xlCellTypeVisible)
                                For Each l In rng
                                   'l.Value = "'" & Trim(.getstring(1, 26, 2)) & "-" & Trim(.getstring(i, 14, 2))
                                    If Format(wS.Range("Y" & l.Row), "0.00") = Trim(.Getstring(i, 40, 11)) Then
                                       'If Len(Trim(.getstring(i + 6, 2, 11))) > 0 Then
                                        wS.Range("AA" & l.Row) = IIf(Len(Trim(.Getstring(i + 6, 2, 11))) < 1, "0.00", Trim(.Getstring(i + 6, 2, 11))) & _
                                        "-" & IIf(Len(Trim(.Getstring(i + 6, 14, 3))) > 0, Trim(.Getstring(i + 6, 14, 3)), "001")
                                    End If
                                Next l
                               'rng.Value = "'" & Trim(.getstring(1, 26, 2)) & "-" & Trim(.getstring(i, 14, 2))
                            End If
                        End If
                    Next i
                Else
                    Exit For
                End If
NxtDraft:
               .SendKeys ("<Pf9>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
                PLACEVALUE CCN, 8, 15
               .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
                p = 1 ' ADDED 2023.08.03
                Do While p < Pg ' ADDED 2023.08.03
                    .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop: p = p + 1 ' ADDED 2023.08.03
                    DoEvents ' ADDED 2023.08.03
                Loop ' ADDED 2023.08.03
            Next r
        Next Pg
'        If InStr(1, Trim(.getstring(20, 2, 60)), "MORE DATA", vbTextCompare) > 0 Then 'REMOVED 2023.08.03
'            .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
'            GoTo ReCountDraft
'        End If
    End With
    With wS
        GetCPS_Discounts = 0
        If .AutoFilterMode Then .AutoFilterMode = False
       .Range("$S$5:$AE" & lstRw).AutoFilter Field:=1, Criteria1:=CCN
        If .AutoFilter.Range.Columns("S").SpecialCells(xlCellTypeVisible).Cells.Count - 1 > 0 Then
            Set rng = .AutoFilter.Range.Columns(5).Offset(1, 0).Resize(lstRw - 5, 1).SpecialCells(xlCellTypeVisible)
            For Each l In rng
                If Len(l.Value) < 1 Then
                    GetCPS_Discounts = GetCPS_Discounts + 1
                End If
            Next l
        End If
        If .AutoFilterMode Then .AutoFilterMode = False
       .Sort.SortFields.Clear
       .Sort.SortFields.Add Key:=.Range("S5" & ":S" & lstRw), SortOn:=xlSortOnValues, Order:=xlAscending, DataOption:=xlSortTextAsNumbers
       .Sort.SortFields.Add Key:=.Range("T5" & ":T" & lstRw), SortOn:=xlSortOnValues, Order:=xlAscending, DataOption:=xlSortTextAsNumbers
       .Sort.SortFields.Add Key:=.Range("U5" & ":U" & lstRw), SortOn:=xlSortOnValues, Order:=xlDescending, DataOption:=xlSortTextAsNumbers
       .Sort.SortFields.Add Key:=.Range("W5" & ":W" & lstRw), SortOn:=xlSortOnValues, Order:=xlDescending, DataOption:=xlSortTextAsNumbers
        With .Sort
            .SetRange wS.Range("S5:AE" & lstRw)
            .Header = xlYes
            .MatchCase = False
            .Orientation = xlTopToBottom
            .SortMethod = xlPinYin
            .Apply
        End With
       .Application.ScreenUpdating = True
    End With
    Set wS = Nothing
   'End
End Function
