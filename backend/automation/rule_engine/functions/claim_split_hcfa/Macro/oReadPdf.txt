Public Sub Medicare_Cob_Information(ByVal pFile As String, ByVal rW As Long, ByVal TotalPage As Integer, ByVal LineNo As String) 'ADDED 2024.01.24
    Dim strMark As String, Page As Integer, B As Double, T As Double, L As Double, R As Double
    With iREADER
        For Page = 1 To TotalPage
            strMark = .TextCoordinates(pFile, "MEDICARE/MEDICAID/COB SUPPORT DOCUMENT", Page)
            If Len(Trim(strMark)) > 0 Then
                lineInfo = Split(.TextCoordinates(pFile, Format(LineNo, "000000"), Page), ",")
                Appvd = Split(.TextCoordinates(pFile, "APPVD", Page), ",")
                Paid = Split(.TextCoordinates(pFile, "PAID", Page), ",")
                B = lineInfo(1): T = Replace(lineInfo(3), "|", "")
                L = Paid(0): R = Appvd(2)
                AppvdInfo = Trim(.ReadPage(pFile, Page, L - 15, B, R, T))
                R = Paid(2)
                PaidInfo = Trim(.ReadPage(pFile, Page, L, B, R, T))
                SVL.Range("Z" & rW) = "'" & Replace(AppvdInfo, "$", "")
                SVL.Range("AA" & rW) = "'" & Replace(PaidInfo, "$", "")
                Exit For
            End If
        Next Page
    End With
objRelease:
End Sub

Public Sub Medicare_Medicaid_Cob_Information(ByVal pFile As String, ByVal rW As Integer) 'ADDED 2025.03.11
    Dim TotalPage As Integer, Page As Integer, strMark As String, i As Integer, B As String, T As String
    Dim KeyPattern(8) As String, olCoordinates As String, olSet() As String, h As Integer, cAdjInfo As String
    Dim tmpVal As String
    KeyPattern(0) = "Deductible"
    KeyPattern(1) = "Coinsurance"
    KeyPattern(2) = "Calculated Approved Amount"
    KeyPattern(3) = "Paid"
    KeyPattern(4) = "Patient Responsibility"
    KeyPattern(5) = "Non-Covered"
    KeyPattern(6) = "Contractual"
    KeyPattern(7) = "Medicare ID"
    KeyPattern(8) = "Other Insurance Type"
    With iREADER
        TotalPage = .TotalPages(pFile)
        For Page = 1 To TotalPage
            strMark = .TextCoordinates(pFile, "MEDICARE/MEDICAID/COB SUPPORT DOCUMENT", Page)
            If Len(Trim(strMark)) > 0 Then
                CurPg = Page: Exit For
            End If
        Next Page
        If CurPg < 1 Then GoTo objRelease
        For i = 0 To UBound(KeyPattern)
            olCoordinates = .TextCoordinates(pFile, KeyPattern(i), CurPg)
            If Len(Trim(olCoordinates)) > 0 Then
                olSet = Split(olCoordinates, "|")
                tmpVal = NORMALIZE_EDIT_MSG(.ReadPage(pFile, CurPg, Round(Split(olSet(0), ",")(0)), Round(Split(olSet(0), ",")(1)), 432, Round(Split(olSet(0), ",")(3))))
                tmpVal = Trim(Replace(Replace(tmpVal, KeyPattern(i), ""), " ", ""))
                'tmpVal = Trim(Right(tmpVal, Len(tmpVal) - InStr(1, tmpVal, "$")))
                Select Case KeyPattern(i)
                    Case "Deductible": INF.Range("BR" & rW) = "'" & tmpVal
                    Case "Coinsurance": INF.Range("BS" & rW) = "'" & tmpVal
                    Case "Calculated Approved Amount": INF.Range("BT" & rW) = "'" & tmpVal
                    Case "Paid": INF.Range("BU" & rW) = "'" & tmpVal
                    Case "Patient Responsibility": INF.Range("BV" & rW) = "'" & tmpVal
                    Case "Non-Covered": INF.Range("BW" & rW) = "'" & tmpVal
                    Case "Contractual": INF.Range("BX" & rW) = "'" & tmpVal
                    Case "Medicare ID": INF.Range("BZ" & rW) = "'" & tmpVal
                    Case "HIC Number": INF.Range("CA" & rW) = "'" & tmpVal
                    Case "Other Insurance Type": INF.Range("CB" & rW) = "'" & tmpVal
                End Select
            End If
        Next i
        strMark = .TextCoordinates(pFile, "SEQ) GRP CD, ADJ RSN: AMT", Page)
        For h = 1 To UBound(Split(strMark, "|"))
            strMark = Split(strMark, "|")(h - 1): B = Round(Split(strMark, ",")(1), 0): T = Round(Split(strMark, ",")(3), 0): cAdjInfo = ""
            If InStr(1, Trim(.ReadPage(pFile, CurPg, 36, B, Round(Split(strMark, ",")(2), 0), T)), "LINE#") > 0 Then 'Added logic 2023.03.14
                For i = 1 To 22 'Table Section Info
                    B = B - 12.44: T = T - 12.44
                    tmpVal = Replace(Trim(.ReadPage(pFile, Page, Round(Split(strMark, ",")(0), 0), B, Round(Split(strMark, ",")(2), 0), T)), " ", "")
                    If Len(Trim(tmpVal)) < 1 Then
                        tmpVal = Replace(Trim(.ReadPage(pFile, Page, Round(Split(strMark, ",")(0), 0), B + 5, Round(Split(strMark, ",")(2), 0), T + 5)), " ", "")
                        If Len(Trim(tmpVal)) > 0 Then
                            cAdjInfo = cAdjInfo & " " & tmpVal
                        End If
                    Else
                        cAdjInfo = cAdjInfo & " " & tmpVal
                    End If: DoEvents
                Next i
            Else
                For i = 1 To 4 'Header Section Info
                    B = B - 9: T = T - 8
                    tmpVal = Replace(Trim(.ReadPage(pFile, Page, Round(Split(strMark, ",")(0), 0), B, Round(Split(strMark, ",")(2), 0), T)), " ", "")
                    If Left(Trim(tmpVal), 2) = i & ")" Then
                        cAdjInfo = cAdjInfo & " " & tmpVal
                    End If: DoEvents
                Next i: DoEvents
            End If
        Next h
        INF.Range("BY" & rW) = "'" & Trim(cAdjInfo)
    End With
objRelease:
End Sub

Public Function Reformat_Address(ByVal StrAdr As String, ByVal rW As Integer, ByVal TypInfo As String, ByRef wsRef As Worksheet)
    Dim AdrArray() As String
    AdrArray = Split(StrAdr, "|") 'Ubound(adrarray) - Lbound(adrarray) + 1
    With wsRef
        Select Case UBound(AdrArray)
            Case 2
                 Select Case TypInfo
                    Case "SERVICE"
                        .Range("AR" & rW) = Trim(AdrArray(0))
                        .Range("AS" & rW) = PoBoxAdr_Collections(Trim(AdrArray(1)))
                        .Range("AT" & rW) = ""
                        .Range("AU" & rW) = Trim(Left(Trim(AdrArray(2)), InStr(1, Trim(AdrArray(2)), ",") - 1))
                         AdrArray(2) = Replace(Replace(AdrArray(2), .Range("AU" & rW), ""), ",", " ")
                        .Range("AV" & rW) = Trim(Mid(Trim(AdrArray(2)), InStr(1, Trim(AdrArray(2)), ",") + 1, 2))
                        .Range("AW" & rW) = "'" & Right(Trim(AdrArray(2)), Len(Trim(AdrArray(2))) - InStr(1, Trim(AdrArray(2)), " "))
                    Case "BILLING"
                        .Range("BA" & rW) = Trim(AdrArray(0))
                        .Range("BB" & rW) = PoBoxAdr_Collections(Trim(AdrArray(1)))
                        .Range("BC" & rW) = ""
                        .Range("BD" & rW) = Trim(Left(Trim(AdrArray(2)), InStr(1, Trim(AdrArray(2)), ",") - 1))
                         AdrArray(2) = Replace(Replace(AdrArray(2), .Range("BD" & rW), ""), ",", " ")
                        .Range("BE" & rW) = Trim(Mid(Trim(AdrArray(2)), InStr(1, Trim(AdrArray(2)), ",") + 1, 2))
                         If Len(Trim(.Range("BE" & rW))) < 2 Then .Range("BE" & rW).ClearContents
                        .Range("BF" & rW) = "'" & Right(Trim(AdrArray(2)), Len(Trim(AdrArray(2))) - InStr(1, Trim(AdrArray(2)), " "))
                 End Select
            Case 3
                 Select Case TypInfo
                    Case "SERVICE"
                        .Range("AR" & rW) = Trim(AdrArray(0))
                        .Range("AS" & rW) = PoBoxAdr_Collections(Trim(AdrArray(1)))
                        .Range("AT" & rW) = PoBoxAdr_Collections(Trim(AdrArray(2)))
                        .Range("AU" & rW) = Trim(Left(Trim(AdrArray(3)), InStr(1, Trim(AdrArray(3)), ",") - 1))
                         AdrArray(3) = Replace(Replace(AdrArray(3), .Range("AU" & rW), ""), ",", " ")
                        .Range("AV" & rW) = Trim(Mid(Trim(AdrArray(3)), InStr(1, Trim(AdrArray(3)), ",") + 1, 2))
                        .Range("AW" & rW) = "'" & Right(Trim(AdrArray(3)), Len(Trim(AdrArray(3))) - InStr(1, Trim(AdrArray(3)), " "))
                    Case "BILLING"
                        .Range("BA" & rW) = Trim(AdrArray(0))
                        .Range("BB" & rW) = PoBoxAdr_Collections(Trim(AdrArray(1)))
                        .Range("BC" & rW) = PoBoxAdr_Collections(Trim(AdrArray(2)))
                        .Range("BD" & rW) = Trim(Left(Trim(AdrArray(3)), InStr(1, Trim(AdrArray(3)), ",") - 1))
                         AdrArray(3) = Replace(Replace(AdrArray(3), .Range("BD" & rW), ""), ",", " ")
                        .Range("BE" & rW) = Trim(Mid(Trim(AdrArray(3)), InStr(1, Trim(AdrArray(3)), ",") + 1, 2))
                        .Range("BF" & rW) = "'" & Right(Trim(AdrArray(3)), Len(Trim(AdrArray(3))) - InStr(1, Trim(AdrArray(3)), " "))
                 End Select
        End Select
    End With
End Function

Public Function PoBoxAdr_Collections(ByVal AdrParam As String) As String
    If Len(Trim(AdrParam)) < 1 Then Exit Function
    Dim PoBoxCol As Dictionary, i As Integer
    Set PoBoxCol = New Dictionary
    PoBoxAdr_Collections = AdrParam
    PoBoxCol.Add "POBOX ", "PO BOX "
    PoBoxCol.Add "P O BOX ", "PO BOX "
    PoBoxCol.Add "P  O BOX ", "PO BOX "
    PoBoxCol.Add "P  O  BOX ", "PO BOX "
    PoBoxCol.Add "P.O.BOX ", "PO BOX "
    PoBoxCol.Add "P. O. BOX ", "PO BOX "
    PoBoxCol.Add "P.O. BOX ", "PO BOX "
    PoBoxCol.Add "P O BOX  ", "PO BOX "
    PoBoxCol.Add "P.O BOX ", "PO BOX "
    PoBoxCol.Add "POST OFFICE BOX ", "PO BOX "
    PoBoxCol.Add "POBOX", "PO BOX "
    PoBoxCol.Add "P O BOX", "PO BOX "
    PoBoxCol.Add "P  O BOX", "PO BOX "
    PoBoxCol.Add "P  O  BOX", "PO BOX "
    PoBoxCol.Add "P.O.BOX", "PO BOX "
    PoBoxCol.Add "P. O. BOX", "PO BOX "
    PoBoxCol.Add "P.O. BOX", "PO BOX "
    PoBoxCol.Add "P.O BOX", "PO BOX "
    PoBoxCol.Add "POST OFFICE BOX", "PO BOX "
    
    For i = 0 To PoBoxCol.Count - 1
        If InStr(1, AdrParam, PoBoxCol.Keys(i), vbTextCompare) > 0 Then
            PoBoxAdr_Collections = Replace(AdrParam, PoBoxCol.Keys(i), PoBoxCol.Items(i))
            Exit For
        End If
    Next i
    
    PoBoxCol.RemoveAll
    Set PoBoxCol = Nothing
End Function
