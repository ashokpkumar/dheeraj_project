
Public Sub GetCPS_Discounts(ByVal CCN As String) 'Added 2023.05.04
    Dim R As Integer, i As Integer, ln As String, rng As Range, lstRw As Long, L As Range, k As Integer
    Dim BgnDt As Date
    If MN.AutoFilterMode Then MN.AutoFilterMode = False
'    MN.Application.ScreenUpdating = False
    lstRw = MN.Cells(MN.Rows.Count, "S").End(xlUp).Row
    With objSession.screen
ReCountDraft:
        For R = 6 To 18 Step 2
            ln = Trim(.GetString(R, 2, 2))
            If Len(ln) > 0 Then
                PLACEVALUE ln, 3, 26
               .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
               .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
                k = 9
                For i = 5 To 11 Step 2
                    If Len(Trim(.GetString(i, 4, 6))) > 0 Then
                        BgnDt = CDate(Trim(.GetString(i, 4, 2)) & "/" & Trim(.GetString(i, 6, 2)) & "/" & Trim(.GetString(i, 8, 2)))
                        MN.Range("$S$5:$AI" & lstRw).AutoFilter Field:=1, Criteria1:=CCN
                        MN.Range("$S$5:$AI" & lstRw).AutoFilter Field:=2, Criteria1:=Format(BgnDt, "MM/DD/YY") 'BGN DATE
                        MN.Range("$S$5:$AI" & lstRw).AutoFilter Field:=4, Criteria1:=Trim(.GetString(i, 41, 5)) 'CPT CODES
                        MN.Range("$S$5:$AI" & lstRw).AutoFilter Field:=14, Criteria1:=Trim(.GetString(i + 1, 70, 6)) 'DX CODES
                        If MN.AutoFilter.Range.Columns("S").SpecialCells(xlCellTypeVisible).Cells.Count - 1 > 0 Then
                            Set rng = MN.AutoFilter.Range.Columns(9).Offset(1, 0).Resize(lstRw - 5, 1).SpecialCells(xlCellTypeVisible)
                            For Each L In rng
                                L.Value = "'" & IIf(Len(Trim(.GetString(i + k, 2, 11))) > 0, Trim(.GetString(i + k, 2, 11)), "0.00") & "-" & _
                                IIf(Len(Trim(.GetString(i + k, 14, 3))) > 0, Trim(.GetString(i + k, 14, 3)), "001") ' & "|" & _
                                Trim (.getstring(i + k, 18, 11)) & "-" & Trim(.getstring(i + k, 30, 3)) '& "|" & _
                                Trim(.getstring(i + k + 6, 4, 11)) & "-" & Trim(.getstring(i + k + 6, 16, 11)) & "-" & _
                                Trim(.getstring(i + k + 6, 28, 1)) & "-" & Trim(.getstring(i + k + 6, 31, 1)) & "-" & _
                                Trim(.getstring(i + k + 6, 36, 11)) & "-" & Trim(.getstring(i + k + 6, 48, 5)) & "-" & _
                                Trim(.getstring(i + k + 6, 56, 11)) 'INEL-CD1|INDEL-CD2|OIELIG-OIPAID-OI-BN-BNAMT-SP-NDC
                            Next L
                        End If
                    End If: k = k - 1
                Next i
            Else
                Exit For
            End If
NxtDraft:
           .SendKeys ("<Pf9>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
            PLACEVALUE CCN, 8, 15
           .SendKeys ("<Enter>"): Do While .OIA.Xstatus <> 0: DoEvents: Loop
        Next R
        If InStr(1, Trim(.GetString(20, 2, 60)), "MORE DATA", vbTextCompare) > 0 Then
            .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
            GoTo ReCountDraft
        End If
    End With
    If MN.AutoFilterMode Then MN.AutoFilterMode = False
End Sub
