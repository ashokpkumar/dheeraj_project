Public Function GetDB(ByVal dbEnvironment As String) As String
   'Set for desired Database name in Data API. Used by other subroutines to build the correct FULL URL for Data API...
    Select Case UCase(Trim(dbEnvironment))
        Case "DEV": GetDB = "mgD"
        Case "TEST": GetDB = "mgT"
        Case "PROD": GetDB = "mgP"
    End Select
End Function

Public Function GetAPIPath(ByVal dbEnvironment As String) As String
   'Set for desired Data API (basic) URL. Used by other subroutines to build the correct FULL URL for Data API...
   'TEST - NOTE: Test & DEV databases both use the TEST API Path
    Select Case UCase(Trim(dbEnvironment))
        Case "DEV", "TEST": GetAPIPath = "http://eutsdotr-test.optum.com/dataapi/api/DataInterop/GetQueryData" 'NOT IN USE
        Case "PROD": GetAPIPath = "https://dataapi.appservice.ctc01.optum.com/api/DataInterop/GetQueryData" 'NEW PATH 07022021
    End Select
End Function

Public Function GetCharArray() As String
   'Necessary environmental settings for the Data API web interface process. Do not alter.
    Dim strArray As String: Dim intStartChar As Integer: Dim intEndChar As Integer: Dim strNextChar As Integer
    GetCharArray = ""
    strArray = "38,97,112,112,107,101,121,61,52,98,52,57,52,99,52,99,50,48,53,51,52,49,53,50,52,49,52,56,50,48,52,51,52,102,52,101,52,101,52,102,53,50,"
    intStartChar = 1
    Do
        intEndChar = InStr(intStartChar, strArray, ",")
        If intEndChar <> 0 Then
            strNextChar = Mid(strArray, intStartChar, intEndChar - intStartChar)
            GetCharArray = GetCharArray & Chr(strNextChar)
            intStartChar = intEndChar + 1
        End If
    Loop While intEndChar <> 0
End Function
