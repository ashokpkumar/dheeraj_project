Public Function Current_Session() As Boolean
    Dim objSystem As Object, objSessions As Object, msg As String
    On Error GoTo ErrHandler
    Current_Session = False
    Set objSystem = CreateObject("EXTRA.System")
    If (objSystem Is Nothing) Then GoTo ErrHandler
    Set objSessions = objSystem.Sessions
    If (objSessions Is Nothing) Then GoTo ErrHandler
    OldSystemTimeout& = objSystem.TimeoutValue: g_HostSettleTime = 500
    OldSystemTimeout& = objSystem.TimeoutValue
    If (g_HostSettleTime > OldSystemTimeout) Then objSystem.TimeoutValue = g_HostSettleTime
    If Len(Trim(Sheet1.Range(s1cell_sessPath))) > 0 Then _
    Set objSess = objSessions.Item(Trim(Sheet1.Range(s1cell_sessPath))) Else Set objSess = objSystem.ActiveSession
    If (objSess Is Nothing) Then GoTo ErrHandler
    Current_Session = True
    Exit Function
ErrHandler:
    MsgBox "Unable to communicate with session." & vbNewLine & vbNewLine & msg & vbNewLine & vbNewLine & _
    Err.Description, vbCritical, ThisWorkbook.Name: Set objSess = Nothing: Set objSystem = Nothing: Set objSessions = Nothing
End Function

Public Sub GridPriceCheck(ByVal CCN As String, ByVal rw As Long, ByVal tDraft As Integer)
    Dim ws As Worksheet, claimType As String, Pg As Integer, dLine As Integer, tPages As Integer
    Dim proc As String, chg As Double, dctr As Integer
    Set ws = ThisWorkbook.Worksheets("MAIN")
    With objSess.Screen
       .SendKeys "<Pf9>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        PlaceValue CCN, 8, 15: RemoveValue 9, 15: RemoveValue 12, 15
       .SendKeys "<Enter>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        Select Case tDraft
            Case 1 To 7: tPages = 1
            Case 8 To 14: tPages = 2
            Case 15 To 21: tPages = 3
            Case 22 To 28: tPages = 4
        End Select
        dctr = 1
        claimType = ""
        ws.Range(s1col_statEdit & rw).ClearContents
        For Pg = 1 To tPages
            For dLine = 1 To 7
                PlaceValue Format(dLine, "00"), 3, 26
               .SendKeys "<Enter>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
                ws.Range(s1col_rel & rw) = Trim(.GetString(13, 27, 2)) 'added 2023.07.13
               .SendKeys "<Enter>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
                Select Case Trim(.GetString(1, 2, 11))
                    Case "CPS450.01"
                        claimType = "HCFA"
                        ws.Range(s1col_oldPos & rw) = Trim(.GetString(2, 6, 3))  'Current POS
                        If ws.Range(s1cell_aplyGridPrc) = "Y" Then
                            For c2 = 5 To 11 Step 2
                                If Len(Trim(.GetString(c2, 4, 6))) = 0 Then Exit For
                                chg = CDbl(Trim(.GetString(c2, 26, 7)))
                                proc = Trim(.GetString(c2, 41, 6))
                                If isCorrectAmount(proc, chg) = False Then
                                    If ws.Range(s1col_statEdit & rw) <> "" Then
                                        ws.Range(s1col_statEdit & rw) = ws.Range(s1col_statEdit & rw) & Chr(10) & _
                                        "PRICE MISMATCH " & proc & "(Pg:" & Pg & " Ln:" & Format(dLine, "00") & ")"
                                    Else
                                        ws.Range(s1col_statEdit & rw) = "PRICE MISMATCH " & proc & "(Pg:" & Pg & " Ln:" & Format(dLine, "00") & ")"
                                    End If
                                End If
                            Next c2
                        Else
                            GoTo WrapUp
                        End If
                    Case "BLX2460.01"
                        claimType = "UB"
                        ws.Range(s1col_oldPos & rw) = Trim(.GetString(1, 26, 3))  'Current POS
                        GoTo WrapUp
                End Select
                dctr = dctr + 1
                If dctr > tDraft Then GoTo WrapUp
               .SendKeys "<Pf9>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
                PlaceValue Trim(ws.Range(s1col_clamNo & rw)), 8, 15
                RemoveValue 9, 15: RemoveValue 12, 15
               .SendKeys "<Enter>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
            Next dLine
           .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        Next Pg
WrapUp:
        ws.Range(s1col_clamTyp & rw) = claimType
       .SendKeys "<Pf8>": Do While .OIA.Xstatus <> 0: DoEvents: Loop 'ADDED OLD PRVCODE 2024.03.21 ****
        PlaceValue "408", 2, 37
       .SendKeys "<ENTER>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        ws.Range(s1col_oldPrvCd & rw) = "'" & Trim(.GetString(3, 48, 1)) '****
       .SendKeys "<Pf9>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
    End With
End Sub

Public Function Is_It_Fully_Paid(rtnMsg, ByVal ByWhat As String) As Boolean
    Dim tDiscount As Double, tCharge As Double, lN As Integer, tPaid As Double, tOiEl As Double, tOiPd As Double
    Dim RejectedInel As New Dictionary, i As Integer, CPTCodes As New Dictionary
    Is_It_Fully_Paid = True: tDiscount = 0: tCharge = 0
    RejectedInel.RemoveAll: CPTCodes.RemoveAll: i = 2
    With ThisWorkbook.Worksheets("DxCodesRef") 'Added 2023.02.10 (as variables)
        Do While Len(Trim(.Range("H" & i))) > 0 'modified 2023.07.18
            RejectedInel.Add CStr(Trim(.Range("H" & i))), i - 2 'modified 2023.07.18
            i = i + 1
        Loop
    End With
    i = 2
    With ThisWorkbook.Worksheets("DxCodesRef") 'ADDED 2026.03.31
        Do While Len(Trim(.Range("X" & i))) > 0
            CPTCodes.Add CStr(Trim(.Range("X" & i))), i - 2
            i = i + 1
        Loop
    End With
    rtnMsg = ""
    With objSess.Screen
       .SendKeys "<Pf8>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        PlaceValue "408", 2, 37
       .SendKeys "<ENTER>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
Compare:
        For lN = 5 To 19 Step 14
            If Len(Trim(.GetString(lN, 26, 9))) < 1 Then Exit For
            tCharge = Trim(.GetString(lN, 26, 9))
            tPaid = Trim(.GetString(lN + 5, 72, 9))
            Select Case ByWhat
                Case "InelCodes"
                    If RejectedInel.Exists(CStr(Trim(.GetString(lN + 1, 10, 3)))) Or RejectedInel.Exists(Trim(.GetString(lN + 1, 22, 3))) Or _
                    RejectedInel.Exists(Trim(.GetString(lN + 1, 34, 3))) Then 'Reject it. Added 11/18/2020
                        rtnMsg = "REJECTED INELCODES ON 100% PD:" & Trim(.GetString(lN + 1, 10, 3)) & _
                        "|" & Trim(.GetString(lN + 1, 22, 3)) & "|" & Trim(.GetString(lN + 1, 34, 3)) 'Added msg 2023.02.10
                        Is_It_Fully_Paid = False
                        GoTo ExtFunc
                    End If
                    tDiscount = 0
                    If (Trim(.GetString(lN + 1, 10, 3)) <> "000" And Trim(.GetString(lN + 1, 22, 3)) <> "000" And _
                    Trim(.GetString(lN + 1, 34, 3)) <> "000") Or (Trim(.GetString(lN + 1, 10, 3)) <> "000" And _
                    Trim(.GetString(lN + 1, 22, 3)) <> "000") Or (Trim(.GetString(lN + 1, 10, 3)) <> "000" And _
                    Trim(.GetString(lN + 1, 34, 3)) <> "000") Or (Trim(.GetString(lN + 1, 22, 3)) <> "000" And _
                    Trim(.GetString(lN + 1, 34, 3)) <> "000") Then 'Reject if more than 1 inelcode. Added 12/03/2020
                        rtnMsg = "MORE THAN 1 INEL CODES"
                        Is_It_Fully_Paid = False
                        GoTo ExtFunc
                    End If
                    If Trim(.GetString(lN + 1, 10, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 4, 9))
                    If Trim(.GetString(lN + 1, 22, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 16, 9))
                    If Trim(.GetString(lN + 1, 34, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 28, 9))
                    If Trim(.GetString(lN + 1, 10, 3)) = "311" Or Trim(.GetString(lN + 1, 22, 3)) = "311" Or _
                    Trim(.GetString(lN + 1, 34, 3)) = "311" Then 'Added 12/03/2020
                        tOiEl = Trim(.GetString(lN + 8, 3, 10))
                        tOiPd = Trim(.GetString(lN + 8, 15, 10))
                        If CStr(tPaid) <> CStr(tOiEl - tOiPd) Then
                            Is_It_Fully_Paid = False
                            GoTo ExtFunc
                        End If
                    Else 'Condition else added 2023.02.10
                        If CStr(tCharge) <> CStr(tDiscount + tPaid) Then 'Applies if not 311 InelCodes
                            Is_It_Fully_Paid = False
                            GoTo ExtFunc
                        End If
                    End If: DoEvents
                Case "CPTCodes" 'ADDED 2026.03.31
                    If CPTCodes.Exists(CStr(Trim(.GetString(lN, 42, 5)))) Then
                        If Trim(.GetString(lN + 1, 10, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 4, 9))
                        If Trim(.GetString(lN + 1, 22, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 16, 9))
                        If Trim(.GetString(lN + 1, 34, 3)) <> "000" Then tDiscount = Trim(.GetString(lN + 2, 28, 9))
                        If CStr(tCharge) <> CStr(tDiscount + tPaid) Then
                            Is_It_Fully_Paid = False
                            GoTo ExtFunc
                        End If
                    End If
            End Select
        Next lN
        If InStr(1, Trim(.GetString(29, 69, 11)), "F11") > 0 Then
           .SendKeys "<Pf11>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
            If Trim(.GetString(31, 2, 60)) = "YOU MAY NOT PAGE FORWARD AT THIS TIME" Then
                GoTo ExtFunc
            End If
            GoTo Compare
        End If
ExtFunc:
       .SendKeys "<Pf8>": Do While .OIA.Xstatus <> 0: DoEvents: Loop
        PlaceValue "506", 2, 37: Do While .OIA.Xstatus <> 0: DoEvents: Loop
       .SendKeys "<ENTER>": Do While .OIA.Xstatus <> 0: DoEvents: Loop

Public Function CheckTimeUnits(mw As Worksheet) As Boolean 'added 05.06.2022
    CheckTimeUnits = True
    For rw = mw.Range(s1cell_start) To mw.Range(s1cell_end)
        myTU = "": myTU = Trim(mw.Range(s1col_tmeUnit & rw))
        If myTU = "" Then
            mw.Range(s1col_tmeUntStat & rw) = "No listed 'TIME UNIT' value."
            CheckTimeUnits = False
            GoTo NxtItem
        End If
        If IsNumeric(myTU) = False Then

            mw.Range(s1col_tmeUntStat & rw) = "Listed 'TIME UNIT' value is a negative number."

            mw.Range(s1col_tmeUnit & rw) = Format(myTU, "0")
        End If
        If myTU > 99 Then
            mw.Range(s1col_tmeUntStat & rw) = "Listed 'TIME UNIT' value exceeds 99."
            CheckTimeUnits = False
        End If: DoEvents
NxtItem:
    Next rw

Public Function Apply_BondClinic(ByVal rw As Integer, lN As Integer) 'added 04.11.2023
    Dim Chrg As Double, InelAmt As Double, Amt997 As Double, z As Integer
    Select Case lN
        Case 5: z = 9
        Case 7: z = 8
        Case 9: z = 7
        Case 11: z = 6
    End Select
    With objSess.Screen
        Chrg = CDbl(Trim(.GetString(lN, 21, 11)))
        InelAmt = CDbl(Trim(.GetString(lN + z, 2, 11)))
        Amt997 = (Chrg - InelAmt)
        If Chrg > AmtCeiling Then
            If AmtCeiling_Applied <> True Then
                Amt997 = Abs((Chrg - InelAmt) - AmtCeiling)
                AmtCeiling_Applied = True
            End If
            RemoveValue lN + z, 18: PlaceValue Amt997, lN + z, 18
            RemoveValue lN + z, 30: PlaceValue "997", lN + z, 30
        Else
            If AmtCeiling_Applied <> True Then
                AmtCeilingRemaining = (Chrg - InelAmt) - AmtCeiling
                AmtCeiling = Abs(AmtCeilingRemaining)
            End If
            If Chrg > 10 Then
                 RemoveValue lN + z, 18: PlaceValue Amt997, lN + z, 18
                 RemoveValue lN + z, 30: PlaceValue "997", lN + z, 30
            Else
                If AmtCeiling_Applied = True Then
                     If Amt997 > 0 Then
                        RemoveValue lN + z, 18: PlaceValue Amt997, lN + z, 18
                        RemoveValue lN + z, 30: PlaceValue "997", lN + z, 30
                    End If
                End If
            End If
        End If
    End With
End Function
